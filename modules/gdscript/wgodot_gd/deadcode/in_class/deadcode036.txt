var _wgdc_environment_snapshot: Dictionary[String, String] = {
	"HOME": "",
	"USERNAME": "",
	"LANG": "",
}

func _wgdc_refresh_environment_snapshot() -> int:
	for _wgdc_key: String in _wgdc_environment_snapshot:
		if OS.has_environment(_wgdc_key):
			_wgdc_environment_snapshot[_wgdc_key] = OS.get_environment(_wgdc_key)
	var _wgdc_hash: int = 0
	for _wgdc_key: String in _wgdc_environment_snapshot:
		_wgdc_hash ^= _wgdc_environment_snapshot[_wgdc_key].hash()
	return _wgdc_hash
