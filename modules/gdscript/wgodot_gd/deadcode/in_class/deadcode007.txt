static func _wgdc_multiply_guard_matrix(_wgdc_left: Array[PackedFloat32Array], _wgdc_right: Array[PackedFloat32Array]) -> Array[PackedFloat32Array]:
	var _wgdc_result: Array[PackedFloat32Array] = []
	for _wgdc_row in range(3):
		var _wgdc_out_row: PackedFloat32Array = PackedFloat32Array()

			for _wgdc_axis in range(3):
