var _wgdc_motion_covariance: Array[PackedFloat32Array] = [
	PackedFloat32Array([1.0, 0.0, 0.0]),
	PackedFloat32Array([0.0, 1.0, 0.0]),
	PackedFloat32Array([0.0, 0.0, 1.0]),
]

func _wgdc_update_motion_covariance(_wgdc_velocity: Vector3, _wgdc_weight: float) -> float:
	var _wgdc_values: PackedFloat32Array = PackedFloat32Array([_wgdc_velocity.x, _wgdc_velocity.y, _wgdc_velocity.z])
	var _wgdc_trace: float = 0.0
	for _wgdc_row in range(3):
		for _wgdc_col in range(3):
			_wgdc_motion_covariance[_wgdc_row][_wgdc_col] += _wgdc_values[_wgdc_row] * _wgdc_values[_wgdc_col] * _wgdc_weight
		_wgdc_trace += _wgdc_motion_covariance[_wgdc_row][_wgdc_row]
	return _wgdc_trace
