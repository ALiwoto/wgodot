static func _wgdc_static_node_path_weight(_wgdc_path: NodePath) -> int:
	var _wgdc_names: PackedStringArray = String(_wgdc_path).split("/", false)
	var _wgdc_weight: int = 0
	for _wgdc_name in _wgdc_names:
		_wgdc_weight += String(_wgdc_name).hash()
	return _wgdc_weight
